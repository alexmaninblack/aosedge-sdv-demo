cmake_minimum_required(VERSION 3.23)
project(cm_storage_restart_proof LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CORE "" CACHE PATH "Pinned aos_core_lib_cpp checkout with the test-only patch")
if(NOT EXISTS "${CORE}/src/core/cm/launcher/CMakeLists.txt")
    message(FATAL_ERROR "Set CORE to the pinned aos_core_lib_cpp checkout")
endif()
set(CMAKE_MODULE_PATH ${CORE}/cmake)
include(AddModule)
find_package(GTest REQUIRED)
include(GoogleTest)
enable_testing()
set(TARGET_INCLUDES ${CORE}/src)
set(HEADERS_BASE_DIRS ${CORE}/src)
set(WITH_TEST OFF)
set(WITH_MBEDTLS OFF CACHE BOOL "")
set(WITH_OPENSSL OFF CACHE BOOL "")
set(TARGET_NAMESPACE aos::core::common)
set(TARGET_PREFIX aos_core_common)
add_subdirectory(${CORE}/src/core/common/crypto crypto)
add_subdirectory(${CORE}/src/core/common/tools tools)
set(TARGET_NAMESPACE aos::core::common::tests)
set(TARGET_PREFIX aos_core_common_tests)
add_subdirectory(${CORE}/src/core/common/tests/utils utils)
set(TARGET_NAMESPACE aos::core::cm)
set(TARGET_PREFIX aos_core_cm)
set(WITH_TEST ON)
add_subdirectory(${CORE}/src/core/cm/launcher launcher)
target_link_libraries(aos_core_cm_launcher_test PRIVATE aos::core::common::tests::utils)
